= Error-Lang Studio: IDE Architecture
:toc:
:sectnums:

== Vision

Error-Lang Studio is a web-based IDE that makes computational forces visible and navigable. It's not just a code editor - it's an **archaeology toolkit** for exploring how code transforms through abstraction layers.

== Architecture: The Elm Architecture (TEA) in ReScript

We use TEA because it's:
- **Predictable**: Unidirectional data flow
- **Debuggable**: Time-travel debugging built-in
- **Testable**: Pure functions everywhere
- **Scalable**: Composable components

=== Core Pattern

```rescript
// The TEA pattern
type model = { /* application state */ }
type msg = Update | Click | ... /* events */

let init = () => (initialModel, Cmd.none)
let update = (msg, model) => (newModel, Cmd.none)
let view = (model) => Html.div(...)
```

== Model: Application State

```rescript
// ide/src/Model.res

type model = {
  // Code state
  code: string,
  cursorPosition: {line: int, column: int},
  selection: option<{start: position, end_: position}>,

  // Compilation state
  ast: option<program>,
  parseErrors: array<diagnostic>,

  // Stability tracking
  stabilityScore: int,
  stabilityHistory: array<int>,
  stabilityFactors: array<stabilityFactor>,

  // Layer navigation

  layerViews: dict<layer, string>,

  // Analysis results
  fiveWhysResults: option<analysisResult>,
  quantumVariables: array<(string, quantumVariable)>,
  paradoxes: array<paradox>,

  // UI state
  activePanel: panel,  // Editor, Dashboard, Navigator, Analyzer
  theme: theme,
  fontSize: int,

  // Execution state
  isRunning: bool,
  runtimeOutput: array<string>,
  executionTrace: array<executionStep>,
}

type panel =
  | EditorPanel
  | DashboardPanel
  | NavigatorPanel
  | AnalyzerPanel
  | ExplorerPanel

type layer =
  | Grammar
  | Parser
  | AST
  | Semantics
  | Runtime

type theme = Light | Dark | HighContrast
```

== Messages: Events

```rescript
// ide/src/Msg.res

type msg =
  // Code editing
  | CodeChanged(string)
  | CursorMoved({line: int, column: int})
  | SelectionChanged({start: position, end_: position})

  // Compilation
  | CompileCode
  | CompilationComplete(result<program, array<diagnostic>>)

  // Execution
  | RunProgram
  | StepProgram
  | PauseProgram
  | StopProgram
  | RuntimeOutput(string)

  // Layer navigation
  | NavigateToLayer(layer)
  | SelectASTNode(string)  // Node ID
  | HighlightCode({start: int, end_: int})

  // Analysis
  | AnalyzeStability
  | RunFiveWhys(stabilityFactor)
  | ExploreParadox(paradox)

  // Visualization
  | ShowRippleEffect({source: int, affected: array<int>})
  | ShowCascade({path: array<int>})
  | AnimateStability(int)  // New score

  // UI
  | SwitchPanel(panel)
  | ChangeTheme(theme)
  | ChangeFontSize(int)
  | ToggleSidebar

  // Persistence
  | SaveCode
  | LoadCode(string)
  | ExportProgram
```

== Update: State Transitions

```rescript
// ide/src/Update.res

let update = (msg: msg, model: model): (model, Cmd.msg) => {
  switch msg {
  | CodeChanged(newCode) =>
    // Update code and trigger background compilation
    let newModel = {...model, code: newCode}
    (newModel, Cmd.msg(CompileCode))

  | CompileCode =>
    // Parse and analyze code
    (model, Cmd.async(() => {
      let result = Parser.parse(model.code)
      CompilationComplete(result)
    }))

  | CompilationComplete(Ok(ast)) =>
    // Compilation succeeded - update AST and run analysis
    let newModel = {
      ...model,
      ast: Some(ast),
      parseErrors: [],
    }
    (newModel, Cmd.msg(AnalyzeStability))

  | AnalyzeStability =>
    // Analyze code for stability and paradoxes
    switch model.ast {
    | Some(ast) =>
      let analysis = Analyzer.analyzeProgram(ast, model.stabilityScore)
      let newModel = {
        ...model,
        stabilityScore: analysis.stabilityReport.score,
        stabilityFactors: analysis.stabilityReport.factors,
        paradoxes: analysis.paradoxes,
      }
      (newModel, Cmd.none)
    | None =>
      (model, Cmd.none)
    }

  | NavigateToLayer(layer) =>
    // Switch to different abstraction layer
    let layerView = LayerNavigator.createLayerView(model.ast, layer)
    let newModel = {
      ...model,
      currentLayer: layer,
      layerViews: Dict.set(model.layerViews, layer, layerView),
    }
    (newModel, Cmd.none)

  | RunFiveWhys(factor) =>
    // Perform Five Whys root cause analysis
    let analysis = FiveWhys.analyze(factor)
    let newModel = {...model, fiveWhysResults: Some(analysis)}
    (newModel, Cmd.none)

  | AnimateStability(newScore) =>
    // Animate stability change
    let history = Array.concat(model.stabilityHistory, [newScore])
    let newModel = {
      ...model,
      stabilityScore: newScore,
      stabilityHistory: history,
    }
    (newModel, Cmd.animation(AnimationFrame))

  | _ =>
    (model, Cmd.none)
  }
}
```

== View: UI Components

```rescript
// ide/src/View.res

let view = (model: model): Html.html<msg> => {
  Html.div(
    [class("error-lang-studio")],
    [
      // Top bar
      viewTopBar(model),

      // Main content area
      Html.div(
        [class("main-content")],
        [
          // Left sidebar - layer navigator
          viewLayerNavigator(model),

          // Center - code editor
          viewEditor(model),

          // Right sidebar - stability dashboard
          viewStabilityDashboard(model),
        ]
      ),

      // Bottom panel - output/analysis
      viewBottomPanel(model),
    ]
  )
}

// Top bar with controls
let viewTopBar = (model: model): Html.html<msg> => {
  Html.div(
    [class("top-bar")],
    [
      Html.h1([], [Html.text("Error-Lang Studio")]),

      // Panel switcher
      viewPanelTabs(model),

      // Actions
      Html.div(
        [class("actions")],
        [
          Html.button(
            [onClick(RunProgram), disabled(model.isRunning)],
            [Html.text("▶ Run")]
          ),
          Html.button(
            [onClick(StepProgram)],
            [Html.text("⏭ Step")]
          ),
          Html.button(
            [onClick(AnalyzeStability)],
            [Html.text("🔍 Analyze")]
          ),
        ]
      ),
    ]
  )
}

// Layer navigator (left sidebar)
let viewLayerNavigator = (model: model): Html.html<msg> => {
  Html.div(
    [class("layer-navigator")],
    [
      Html.h2([], [Html.text("Layers")]),

      // Layer stack
      Html.div(
        [class("layer-stack")],
        [
          viewLayerButton(Grammar, model.currentLayer),
          viewLayerArrow(),
          viewLayerButton(Parser, model.currentLayer),
          viewLayerArrow(),
          viewLayerButton(AST, model.currentLayer),
          viewLayerArrow(),
          viewLayerButton(Semantics, model.currentLayer),
          viewLayerArrow(),
          viewLayerButton(Runtime, model.currentLayer),
        ]
      ),

      // Current layer view
      viewCurrentLayerDetail(model),
    ]
  )
}

// Code editor (center)
let viewEditor = (model: model): Html.html<msg> => {
  Html.div(
    [class("editor-container")],
    [
      // Monaco editor or CodeMirror
      Html.div(
        [id("editor"), onInput(CodeChanged)],
        []
      ),

      // Inline decorations (paradoxes, warnings)
      ...viewInlineDecorations(model),
    ]
  )
}

// Stability dashboard (right sidebar)
let viewStabilityDashboard = (model: model): Html.html<msg> => {
  Html.div(
    [class("stability-dashboard")],
    [
      Html.h2([], [Html.text("Stability")]),

      // Animated stability bar
      viewStabilityBar(model.stabilityScore),

      // Sparkline trend
      viewStabilityTrend(model.stabilityHistory),

      // Breakdown by category
      Html.div(
        [class("stability-breakdown")],
        viewStabilityBreakdown(model.stabilityFactors)
      ),

      // Recommendations
      Html.div(
        [class("recommendations")],
        viewRecommendations(model.stabilityFactors)
      ),
    ]
  )
}

// Bottom panel (output/analysis)
let viewBottomPanel = (model: model): Html.html<msg> => {
  Html.div(
    [class("bottom-panel")],
    [
      switch model.activePanel {
      | EditorPanel => viewOutput(model.runtimeOutput)
      | AnalyzerPanel => viewFiveWhysPanel(model.fiveWhysResults)
      | ExplorerPanel => viewParadoxExplorer(model.paradoxes)
      | _ => Html.div([], [])
      }
    ]
  )
}
```
